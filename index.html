// تابع دریافت و نمایش نوبت‌های کاربر در مینی‌اپ async function loadUserBookings(userChatId) { try { const response = await fetch('https://YOUR_WORKER_SUBDOMAIN.workers.dev/admin-data'); const data = await response.json(); const myBookings = (data.bookingsList || []).filter(b => b.userChatId.toString() === userChatId.toString()); const container = document.getElementById('my-bookings-container'); if (!container) return; container.innerHTML = ''; if (myBookings.length === 0) { container.innerHTML = '

شما هیچ نوبت فعالی ندارید.

'; return; } myBookings.forEach(booking => { const div = document.createElement('div'); div.className = 'booking-card'; div.innerHTML = `

خدمت: ${booking.serviceName || 'رزرو نوبت'}

تاریخ: ${booking.date}

ساعت: ${booking.time}

`; container.appendChild(div); }); } catch (e) { console.error("خطا در بارگذاری نوبت‌ها", e); } } // تابع ارسال درخواست لغو نوبت به سرور async function cancelUserBooking(bookingId, userChatId) { if (!confirm('آیا از لغو این نوبت اطمینان دارید؟ (توجه: لغو نوبت فقط تا ۴ ساعت قبل از زمان رزرو امکان‌پذیر است)')) return; try { const response = await fetch('https://YOUR_WORKER_SUBDOMAIN.workers.dev/cancel-booking', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ bookingId, userChatId }) }); const result = await response.json(); if (result.success) { alert('نوبت شما با موفقیت لغو شد.'); loadUserBookings(userChatId); // به‌روزرسانی لیست } else { alert(result.error || 'خطایی در لغو نوبت رخ داد.'); } } catch (e) { alert('خطا در ارتباط با سرور.'); } }
